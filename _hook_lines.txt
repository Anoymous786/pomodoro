85	/**
86	 * Custom hook that owns the entire Pomodoro app state:
87	 * timer, tasks, daily stats, roadmaps, and badges. All state is
88	 * persisted to localStorage via the shared schema in lib/types.ts.
89	 *
90	 * The timer survives tab backgrounding because we store an
91	 * absolute endTimestamp and derive remaining time from it,
92	 * rather than counting setInterval ticks.
93	 *
94	 * Task-driven timing: when a task is selected/started, its
95	 * effective minutes (manual override ?? AI estimate) become the
96	 * work countdown duration. When a task's timer finishes, the task
97	 * is marked complete and the timer auto-advances to the next
98	 * incomplete task in that day's queue after a short break.
99	 */
100	export function usePomodoro() {
101	  // HYDRATION SAFETY (root cause of the "stuck 1-error" pill on load):
102	  // This state was previously initialized with useState(() => loadState()),
103	  // which reads localStorage SYNCHRONOUSLY during the first client render.
104	  // The server/prerender always rendered the EMPTY default state, so any
105	  // saved data (tasks, roadmaps, badges, timer progress) made the hydrated
106	  // tree diverge from the prerendered HTML -> React logged a hydration
107	  // mismatch error on every page load, which the Next.js dev overlay showed
108	  // as a persistent error badge in the corner.
109	  //
110	  // Fix: render deterministically from the empty state first, then adopt
111	  // persisted state in a post-mount effect (standard two-pass pattern).
112	  const [state, setState] = useState<PomodoroAppState>(() =>
113	    createEmptyState(),
114	  );
115	  /** True once persisted localStorage state has been adopted post-mount. */
116	  const [hydrated, setHydrated] = useState(false);
117	  const [now, setNow] = useState<number>(() => Date.now());
118	  /**
119	   * Id of the task whose exam modal is currently open (null = closed).
120	   * UI-only state; the exam itself + result live on the Task in localStorage.
121	   */
122	  const [examTaskId, setExamTaskId] = useState<string | null>(null);
123	  /**
124	   * §11: the per-step check-in currently waiting to be answered. When a step is
125	   * completed (via timer finish or an explicit tick) the UI shows a shared
126	   * single-question modal; the step is only marked done once answered.
127	   */
128	  const [pendingCheckIn, setPendingCheckIn] = useState<{
129	    taskId: string;
130	    stepId: string;
131	    stepTitle: string;
132	    topic: string;
133	    completedBy: "timer" | "manual";
134	  } | null>(null);
135	
136	  /**
137	      * §12: when every step of a roadmap is genuinely complete, a FULL-ROADMAP
138	   * mock exam (~20 MCQ) is pending. The workspace renders a modal to run it.
139	   */
140	  const [pendingRoadmapExam, setPendingRoadmapExam] = useState<null>(null);
141	
142	  /**
143	   * PART 2 §2/§3: a Day Assessment (10-12 MCQs) is pending for this day.
144	   * Surfaced when a day's last step completes; the next day stays LOCKED
145	   * (server-authoritatively) until this is submitted.
146	   */
147	  const [dayAssessmentPending, setDayAssessmentPending] = useState<{
148	    taskId: string;
149	    dayNumber: number;
150	  } | null>(null);
151	
152	  /**
153	   * PART 2 §7/§8: the currently active exam (Day Assessment or Final Exam)
154	   * driving the exam section. Draft answers persist via task.assessment.
155	   */
156	  const [activeExam, setActiveExam] = useState<{
157	    taskId: string;
158	    taskName: string;
159	    /** "day" | "final" */
160	    kind: "day" | "final";
161	    dayNumber?: number;
162	    questions: RoadmapExamQuestion[];
163	    /** PART 2 §8: persisted draft restored on resume — answers + position. */
164	    draftAnswers?: Record<string, string>;
165	    draftCurrent?: number;
166	  } | null>(null);
167	
168	
169	
170	  // (Phase 4) Focus tracking — refs (not state) because they are written by
171	  // DOM/visibility events and read synchronously at session end, so they never
172	  // trigger re-renders. Tracking is cleared whenever there is no active work
173	  // session, and accumulated during a hidden-while-work-running spell.
174	  const focusSessionStartRef = useRef<number | null>(null);
175	  const focusHiddenStartRef = useRef<number | null>(null);
176	  const focusHiddenMsRef = useRef(0);
177	  const focusHiddenCountRef = useRef(0);
178	  const lastFocusResultRef = useRef<{
179	    focusPercent: number;
180	    distractions: number;
181	    distractionSeconds: number;
182	  } | null>(null);
183	
184	  // Pass 2 (client-only): adopt persisted state AFTER the first paint.
185	  // Declared BEFORE the persist effect so the very first commit only LOADS,
186	  // never saves.
187	  useEffect(() => {
188	    setState(loadState());
189	    setHydrated(true);
190	    // §13 (item 5): when pointed at the deployed stack, pull the user's
191	    // DURABLE per-topic skill profile from the skill-profile Lambda
192	    // (DynamoDB `roadmaps` table, SKILL# item kind) and merge it over
193	    // anything stale in localStorage — scores persist across sessions
194	    // and devices, not just per-browser localStorage.
195	    if (isRemoteApi()) {
196	      apiGet("/api/skill-profile")
197	        .then(async (res) => {
198	          if (!res.ok) return;
199	          // The skill-profile Lambda returns { profiles: SkillProfile[] }.
200	          const data = (await res.json()) as { profiles?: SkillProfile[] };
201	          const remote = data.profiles;
202	          if (!Array.isArray(remote)) return;
203	          setState((prev) => {
204	            const local = prev.skillProfiles ?? {};
205	            const merged: Record<string, SkillProfile> = { ...local };
206	            let changed = false;
207	            for (const p of remote) {
208	              if (!p || typeof p !== "object" || !p.topic) continue;
209	              const cur = local[p.topic];
210	              // Newest write wins per topic; remote is the durable source.
211	              if (cur && cur.lastUpdated >= (p.lastUpdated ?? 0)) continue;
212	              merged[p.topic] = { ...p, lastUpdated: p.lastUpdated ?? 0 };
213	              changed = true;
214	            }
215	            if (!changed) return prev;
216	            return { ...prev, skillProfiles: merged };
217	          });
218	        })
219	        .catch(() => undefined); // best-effort; offline keeps local data
220	    }
221	  }, []);
222	
223	  // Persist state to localStorage whenever it changes — but only after
224	  // hydration. Without this guard, the first commit (still holding the
225	  // empty default state) would overwrite the user's saved data before it
226	  // was ever loaded.
227	  useEffect(() => {
228	    if (!hydrated) return;
229	    saveState(state);
230	  }, [state, hydrated]);
231	
232	  // ===================================================================
233	  // PHASE 2 — DURABLE PROGRESS SYNC (server = source of truth)
234	  // ===================================================================
235	  /** Latest practiceTasks for timers/callbacks (refs never go stale). */
236	  const practiceTasksRef = useRef(state.practiceTasks);
237	  practiceTasksRef.current = state.practiceTasks;
238	
239	  /**
240	   * Per-task sync bookkeeping: last signature pushed/acknowledged and the
241	   * rev we believe the server holds. Kept OUT of React state so pushes
242	   * never re-trigger the effect that schedules them (no loops).
243	   */
244	  const sigRef = useRef<Map<string, string>>(new Map());
245	  const revRef = useRef<Map<string, number>>(new Map());
246	  /** taskId -> timeout for coalescing bursts into one push. */
247	  const pushTimerRef = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());
248	  /** taskId -> true while a pushTaskNow is awaiting (prevents two pushes
249	   *  for the same task from racing on the same expectedRev, which would
250	   *  otherwise produce a spurious 409 that could re-adopt a stale doc). */
251	  const inFlightPushRef = useRef<Map<string, boolean>>(new Map());
252	  /** True once the initial server hydration has completed. */
253	  const remoteHydratedRef = useRef(false);
254	
255	  /** Coalesced push: at most one PUT per task per ~1.5s burst window. */
256	  const schedulePush = useCallback((taskId: string, delay = 1500) => {
257	    const timers = pushTimerRef.current;
258	    if (timers.has(taskId)) return; // already scheduled
259	    const t = setTimeout(() => {
260	      timers.delete(taskId);
261	      void pushTaskNow(taskId);
262	    }, delay);
263	    timers.set(taskId, t);
264	  }, []);
265	
266	  /**
267	   * Fire-and-forget conditional write of ONE task's progress doc.
268	   * On 409 the authoritative doc is re-fetched; if the server is genuinely
269	   * ahead we adopt it (except a locally running timer keeps its seconds),
270	   * otherwise a concurrent same-device push that won with a STALER snapshot
271	   * must NOT wipe richer local data — the local doc wins and is re-pushed
272	   * from the server's baseline rev (conflict-free upgrade).
273	   */
274	  const pushTaskNow = useCallback(async (taskId: string) => {
275	    if (!remoteHydratedRef.current) return;
276	    const inFlight = inFlightPushRef.current;
277	    if (inFlight.get(taskId)) {
278	      // Another push for this task is awaiting; a 2nd one with the same
279	      // expectedRev would race it into a spurious 409. Re-arm a short
280	      // timer instead — the change-driven effect re-fires once state lands.
281	      schedulePush(taskId, 300);
282	      return;
283	    }
284	    const task = practiceTasksRef.current.find((t) => t.id === taskId);
285	    if (!task) return;
286	    const payload = buildProgressDocPayload(task);
287	    const sig = progressSignature({ ...task, rev: undefined });
288	    if (sigRef.current.get(taskId) === sig) return; // nothing durable changed
289	    const expectedRev = revRef.current.get(taskId) ?? 0;
290	    inFlight.set(taskId, true);
291	    let res;
292	    try {
293	      res = await putProgressDoc(
294	        { taskId, rev: expectedRev, task: payload },
295	        expectedRev
296	      );
297	    } finally {
298	      inFlight.delete(taskId);
299	    }
300	    if (res.ok && res.doc) {
301	      revRef.current.set(taskId, res.doc.rev);
302	      sigRef.current.set(taskId, sig);
303	      // Stamp the acknowledged rev back onto the task in state so the
304	      // persisted localStorage copy tracks the server. Otherwise a reload
305	      // would re-hydrate with a STALE rev (the original adoption rev), the
306	      // remote doc's higher rev would win, and the server's older snapshot
307	      // could overwrite fresher local progress that the last push hadn't
308	      // yet landed (e.g. mid-run seconds rounded to 2s vs the final 7s).
309	      setState((prev) => ({
310	        ...prev,
311	        practiceTasks: prev.practiceTasks.map((t) =>
312	          t.id === taskId
313	            ? {
314	                ...t,
315	                rev: res.doc?.rev ?? t.rev,
316	                syncedAt: Date.now(),
317	              }
318	            : t,
319	        ),
320	      }));
321	      return;
322	    }
323	    if (res.conflict) {
324	      // A concurrent write advanced the doc first. Re-fetch the authoritative
325	      // version. If the server is genuinely ahead (higher rev, or a richer
326	      // snapshot) we adopt it; but if its doc is STALER than our local copy
327	      // (same-device concurrent push won with an older snapshot), keep local
328	      // and re-push from the server's baseline rev — never wipe richer data.
329	      const docs = await fetchProgressDocs();
330	      const remote = (docs ?? []).find((d) => d.taskId === taskId);
331	      const remoteTask = remote ? taskFromProgressDoc(remote) : null;
332	      const local = practiceTasksRef.current.find((t) => t.id === taskId) ?? task;
333	      if (!remoteTask) {
334	        revRef.current.delete(taskId); // gone: re-create on next push
335	        sigRef.current.delete(taskId);
336	        schedulePush(taskId, 300);
337	        return;
338	      }
339	      const resolved = mergeHydratedTask(local, remoteTask, remoteTask.rev ?? 1);
340	      const merged = resolved.task;
341	      // Record the SERVER's signature so an unchanged local copy isn't
342	      // re-pushed endlessly; only re-push when the winner is richer local.
343	      sigRef.current.set(
344	        taskId,
345	        progressSignature({ ...(remoteTask as PracticeTask), rev: undefined })
346	      );
347	      revRef.current.set(taskId, resolved.baselineRev);
348	      setState((prev) => ({
349	        ...prev,
350	        practiceTasks: prev.practiceTasks.map((t) =>
351	          t.id === taskId ? merged : t
352	        ),
353	      }));
354	      // If the resolved winner is our richer LOCAL copy, push it up from
355	      // the server baseline (rev already seeded) so the server converges.
356	      if (resolved.task === local) schedulePush(taskId, 300);
357	      return;
358	    }
359	    // Other failures (offline/5xx): signature untouched -> retried later.
360	  }, []);
361	
362	  // Initial server hydration — runs ONCE, after localStorage adoption.
363	  // Remote docs with a higher rev than the local cache are authoritative
364	  // (localStorage is only an offline/UI cache); local-only tasks are
365	  // pushed up so no data is stranded per-device.
366	  useEffect(() => {
367	    if (!hydrated || remoteHydratedRef.current) return;
368	    let cancelled = false;
369	    (async () => {
370	      const docs = await fetchProgressDocs();
371	      if (cancelled) return;
372	      const remoteByTask = new Map<string, ProgressDoc>();
373	      for (const d of docs ?? []) {
374	        if (d && typeof d.taskId === "string") remoteByTask.set(d.taskId, d);
375	      }
376	      setState((prev) => {
377	        const localByTask = new Map(prev.practiceTasks.map((t) => [t.id, t]));
378	        const merged: PracticeTask[] = [];
379	        const seen = new Set<string>();
380	        remoteByTask.forEach((doc) => {
381	          const rt = taskFromProgressDoc(doc);
382	          if (!rt) return;
383	          const local = localByTask.get(doc.taskId);
384	          const resolved = mergeHydratedTask(local, rt, doc.rev ?? 1);
385	          merged.push(resolved.task);
386	          seen.add(doc.taskId);
387	          revRef.current.set(doc.taskId, resolved.baselineRev);
388	          // Mark adopted docs as already-synced so they are NOT pushed
389	          // back (which would overwrite server data with stale local).
390	          sigRef.current.set(
391	            doc.taskId,
392	            progressSignature({ ...rt, rev: undefined })
393	          );
394	        });
395	        // Local-only tasks are deliberately left OUT of sigRef so the
396	        // change-driven effect below schedules their first push (rev 0
397	        // create) once the merged state has rendered.
398	        for (const t of prev.practiceTasks) {
399	          if (!seen.has(t.id)) merged.push(t); // local-only task
400	        }
401	        return { ...prev, practiceTasks: merged };
402	      });
403	      remoteHydratedRef.current = true;
404	    })().catch(() => {
405	      // Offline / endpoint unavailable: stay on the localStorage cache.
406	      remoteHydratedRef.current = true;
407	    });
408	    return () => {
409	      cancelled = true;
410	    };
411	  }, [hydrated, pushTaskNow]);
412	
413	  /**
414	   * Change-driven sync: after every meaningful state change, compare each
415	   * task's durable signature against the last pushed one and schedule a
416	   * coalesced push. This single effect covers ALL required persistence
417	   * moments: timer events (practicedSeconds/startedAt/timerStatus), step
418	   * completion (isDone), question submission (step.checkIn), day
419	   * assessment results (examResult / timingRecord on the steps), day
420	   * unlock (dayNumber changes), and task changes (name/dates/duration/
421	   * roadmap edits). Pure-UI state (e.g. the 500ms clock tick) does not
422	   * change the signature, so it never triggers a push.
423	   */
424	  useEffect(() => {
425	    if (!hydrated) return;
426	    if (!remoteHydratedRef.current) return; // wait for hydrate
427	    for (const t of state.practiceTasks) {
428	      const sig = progressSignature({ ...t, rev: undefined });
429	      if (sigRef.current.get(t.id) !== sig) {
430	        schedulePush(t.id);
431	      }
432	    }
433	  }, [state.practiceTasks, hydrated, schedulePush]);
434	
435	  // Tick every 500ms to re-render the countdown. The actual
436	  // remaining time is always derived from endTime, so this
437	  // interval is only for UI refresh — it never accumulates drift.
438	  useEffect(() => {
439	    const id = window.setInterval(() => setNow(Date.now()), 500);
440	    return () => window.clearInterval(id);
441	  }, []);
442	
443	  // When the timer is running and the end timestamp passes,
444	  // complete the session.
445	  useEffect(() => {
446	    const { timer } = state;
447	    if (!timer.isRunning || timer.endTimestamp === null) return;
448	    if (now >= timer.endTimestamp) {
449	      completeSession();
450	    }
451	  }, [now, state.timer.isRunning, state.timer.endTimestamp]); // eslint-disable-line react-hooks/exhaustive-deps
452	
453	  // (Phase 4) Live focus score — track how much of the active work session is
454	  // spent with the tab hidden/backgrounded using the Page Visibility API.
455	  // Detection is 100% client-side (no backend call); the score is persisted to
456	  // DynamoDB at session end via completeSession. Refs rather than state so the
457	  // DOM events never force re-renders.
458	  useEffect(() => {
459	    const activeWork = state.timer.mode === "work" && state.timer.isRunning;
460	    if (!activeWork) {
461	      // No active work session: clear accumulated focus tracking so the next
462	      // session records only its OWN distractions.
463	      focusSessionStartRef.current = null;
464	      focusHiddenStartRef.current = null;
465	      focusHiddenMsRef.current = 0;
466	      focusHiddenCountRef.current = 0;
467	      return;
468	    }
469	
470	    // A work session is running. If we are just now entering one, snapshot its
471	    // start time; also seed "hidden at session start" so a user who begins the
472	    // session and immediately tabs away is accounted for.
473	    if (focusSessionStartRef.current === null) {
474	      focusSessionStartRef.current = Date.now();
475	      if (
476	        typeof document !== "undefined" &&
477	        document.visibilityState === "hidden"
478	      ) {
479	        focusHiddenStartRef.current = Date.now();
480	      }
481	    }
482	
483	    const onVisibility = () => {
484	      if (typeof document === "undefined") return;
485	      if (document.visibilityState === "hidden") {
486	        if (focusHiddenStartRef.current === null) {
487	          focusHiddenStartRef.current = Date.now();
488	        }
489	      } else {
490	        if (focusHiddenStartRef.current !== null) {
491	          focusHiddenMsRef.current += Date.now() - focusHiddenStartRef.current;
492	          focusHiddenStartRef.current = null;
493	          focusHiddenCountRef.current += 1;
494	        }
495	      }
496	    };
497	
498	    document.addEventListener("visibilitychange", onVisibility);
499	    return () => document.removeEventListener("visibilitychange", onVisibility);
500	    // eslint-disable-next-line react-hooks/exhaustive-deps
501	  }, [state.timer.mode, state.timer.isRunning]);
502	
503	  // ---- Derived values ----
504	
505	  const settings = state.settings;
506	  const timer = state.timer;
507	  const tasks = state.tasks;
508	
509	  const durationMs = getTimerDurationMs(timer, settings);
510	  // BUG #1 fix: a paused countdown must FREEZE on the preserved `pausedMs`
511	  // (the exact remaining captured at pause time) and NOT keep draining
512	  // against the stale `endTimestamp`. The `now` interval ticks in the
513	  // background regardless of running state (see the interval below), so
514	  // without this guard the dashboard display silently counts down while
515	  // paused. startTimer/startTask rebuild endTimestamp = now + pausedMs on
516	  // resume, so the frozen slice is honored. completeSession/null endTimestamp
517	  // + null pausedMs leaves the idle state showing the full duration.
518	  const pausedRemaining = !timer.isRunning ? (timer.pausedMs ?? 0) : 0;
519	  const remainingMs =
520	    pausedRemaining > 0
521	      ? Math.max(0, pausedRemaining)
522	      : timer.endTimestamp
523	        ? Math.max(0, timer.endTimestamp - now)
524	        : durationMs;
525	  const remainingSeconds = Math.ceil(remainingMs / 1000);
526	  const minutes = Math.floor(remainingSeconds / 60);
527	  const seconds = remainingSeconds % 60;
528	  const displayTime = `${String(minutes).padStart(2, "0")}:${String(
529	    seconds,
530	  ).padStart(2, "0")}`;
531	
532	  const tasksCompleted = countCompletedTasks(tasks);
533	  const totalTasks = tasks.length;
534	
535	  const activeTask = timer.activeTaskId
536	    ? (tasks.find((t) => t.id === timer.activeTaskId) ?? null)
537	    : null;
538	
539	  // Today's stats (from history or computed live).
540	  const todayKey = getTodayKey();
541	  const todayStats: DayStats = state.history[todayKey] ?? {
542	    date: todayKey,
543	    sessions_count: 0,
544	    focus_minutes: 0,
545	    tasks_completed: 0,
546	    total_tasks: 0,
547	    productivity: 0,
548	  };
549	
550	  // Live productivity for today (recomputed on every render so the
551	  // dashboard stays current without waiting for a session to end).
552	  const liveProductivity = computeProductivity(
553	    tasksCompleted,
554	    totalTasks,
555	    todayStats.sessions_count,
556	    settings.dailySessionGoal,
557	  );
558	
559	  // ---- Timer actions ----
560	
561	  const requestNotificationPermission = useCallback(async () => {
562	    if (typeof window !== "undefined" && "Notification" in window) {
563	      if (Notification.permission === "default") {
564	        try {
565	          await Notification.requestPermission();
566	        } catch {
567	          // Permission request failed; ignore.
568	        }
569	      }
570	    }
571	  }, []);
572	
573	  const playChime = useCallback(() => {
574	    try {
575	      const AudioCtx =
576	        window.AudioContext ||
577	        (window as unknown as { webkitAudioContext: typeof AudioContext })
578	          .webkitAudioContext;
579	      const ctx = new AudioCtx();
580	      const playTone = (freq: number, start: number, duration: number) => {
581	        const osc = ctx.createOscillator();
582	        const gain = ctx.createGain();
583	        osc.type = "sine";
584	        osc.frequency.value = freq;
585	        gain.gain.setValueAtTime(0.001, ctx.currentTime + start);
586	        gain.gain.exponentialRampToValueAtTime(
587	          0.3,
588	          ctx.currentTime + start + 0.02,
